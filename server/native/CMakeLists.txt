cmake_minimum_required(VERSION 3.22)
project(zombie_zone_dedicated LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(NATIVE "${CMAKE_CURRENT_LIST_DIR}/../../app/src/main/cpp")
set(MISSION
    "${NATIVE}/world/competitive_map.cpp" "${NATIVE}/world/island.cpp"
    "${NATIVE}/world/structures.cpp" "${NATIVE}/world/buildings.cpp"
    "${NATIVE}/world/world.cpp" "${NATIVE}/world/horizonte_adapter.cpp" "${NATIVE}/world/terrain.cpp"
    "${NATIVE}/world/horizonte/block_mapping.cpp" "${NATIVE}/world/horizonte/terrain.cpp"
    "${NATIVE}/world/horizonte/terrain_noise.cpp" "${NATIVE}/world/horizonte/chunk_generator.cpp"
    "${NATIVE}/world/horizonte/village_generator.cpp")
add_library(mission_map STATIC ${MISSION})
set_target_properties(mission_map PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mission_map PUBLIC "${NATIVE}")
target_compile_definitions(mission_map PUBLIC ZZ_PERF_HOST)
target_compile_options(mission_map PRIVATE -O2 -ffp-contract=off)
add_library(zzbr MODULE addon.cpp "${NATIVE}/game/battle_royale.cpp")
target_link_libraries(zzbr PRIVATE mission_map)
target_include_directories(zzbr PRIVATE "${NATIVE}" /usr/include/node)
target_compile_definitions(zzbr PRIVATE ZZ_PERF_HOST NAPI_VERSION=8 NODE_GYP_MODULE_NAME=zzbr)
target_compile_options(zzbr PRIVATE -O2 -Wall -Wextra -ffp-contract=off)
set_target_properties(zzbr PROPERTIES PREFIX "" SUFFIX ".node")
add_executable(bake_map "${CMAKE_CURRENT_LIST_DIR}/../../tools/bake_competitive_map.cpp")
target_link_libraries(bake_map PRIVATE mission_map)
target_include_directories(bake_map PRIVATE "${NATIVE}")
target_compile_options(bake_map PRIVATE -O2 -ffp-contract=off)
enable_testing()
foreach(name competitive_map competitive_mission_map competitive_wire mission_movement)
    add_executable(${name}_tests "${CMAKE_CURRENT_LIST_DIR}/../../tests/native/${name}_tests.cpp")
    target_link_libraries(${name}_tests PRIVATE mission_map)
    target_compile_options(${name}_tests PRIVATE -UNDEBUG)
    add_test(NAME ${name} COMMAND ${name}_tests)
endforeach()
add_executable(battle_royale_tests "${CMAKE_CURRENT_LIST_DIR}/../../tests/native/battle_royale_tests.cpp" "${NATIVE}/game/battle_royale.cpp")
target_link_libraries(battle_royale_tests PRIVATE mission_map)
target_compile_options(battle_royale_tests PRIVATE -UNDEBUG)
add_test(NAME battle_royale COMMAND battle_royale_tests)
add_executable(character_animation_tests "${CMAKE_CURRENT_LIST_DIR}/../../tests/native/character_animation_tests.cpp" "${NATIVE}/render/character.cpp")
target_include_directories(character_animation_tests PRIVATE "${NATIVE}")
target_compile_options(character_animation_tests PRIVATE -UNDEBUG)
add_test(NAME character_animation COMMAND character_animation_tests)
add_executable(grounded_terrain_tests "${CMAKE_CURRENT_LIST_DIR}/../../tests/native/grounded_terrain_tests.cpp")
target_link_libraries(grounded_terrain_tests PRIVATE mission_map)
target_compile_options(grounded_terrain_tests PRIVATE -UNDEBUG)
add_test(NAME grounded_terrain COMMAND grounded_terrain_tests)
add_executable(terrain_tests "${CMAKE_CURRENT_LIST_DIR}/../../tests/native/terrain_tests.cpp")
target_link_libraries(terrain_tests PRIVATE mission_map)
target_compile_options(terrain_tests PRIVATE -UNDEBUG)
add_test(NAME terrain COMMAND terrain_tests)
add_executable(preload_audit "${CMAKE_CURRENT_LIST_DIR}/../../tools/preload_competitive_map.cpp")
target_link_libraries(preload_audit PRIVATE mission_map)
target_compile_options(preload_audit PRIVATE -O2 -UNDEBUG)
